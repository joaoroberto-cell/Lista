namespace Lista
{
    internal class Program
    {
        static void Main(string[] args)
        {
            List<string> Bancos = new List<string> { "Banco do Brasil", "Caixa", "Santander" };
            Bancos.Add("Nubank");
            Bancos.Insert(2, "Bradesco");

            Bancos.Reverse();
            Bancos.Count();
            Bancos.RemoveAt(Bancos.Count() - 1);
            Bancos.RemoveAt(3);
            Bancos.Clear();
        }
    }
